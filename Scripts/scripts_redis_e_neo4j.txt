ANEXO B: IMPLEMENTAÇÃO REDIS (BANCO CHAVE-VALOR)
Objetivo: Gerenciar sessões de usuários e carrinhos de empréstimo 
com latência sub-milissegundo (dados em memória RAM).

--------------------------------------------------------------------

# 1. Criando o Perfil de Sessão do Usuário (Hash)
# Armazena os dados do leitor para validação rápida de acesso e limites.
HSET sessao:user_9921 nome "Gustavo Lins" status "ATIVO" limite_emprestimos 3 multas 0.0 curso "DSM"

# 2. Adicionando um livro ao Carrinho de Empréstimos (Lista)
# O ID do livro (ISBN/AUD) é adicionado à lista de empréstimos do usuário.
LPUSH carrinho:user_9921 "ISBN-9788535914849"
LPUSH carrinho:user_9921 "AUD-998822"

# 3. Definindo o tempo de expiração (TTL)
# O carrinho expira em 24h (86400 segundos) se o usuário não finalizar.
# Isso é crucial para limpar carrinhos abandonados e liberar a memória RAM.
EXPIRE carrinho:user_9921 86400

# 4. Leitura do carrinho no momento do Checkout
# Retorna todos os itens da lista do usuário.
LRANGE carrinho:user_9921 0 -1


--------------------------------------------------------------------
ANEXO C: IMPLEMENTAÇÃO NEO4J (BANCO DE GRAFOS)
Objetivo: Gerenciar a rede de recomendações de obras baseada em 
relacionamentos complexos (Quem leu X também leu Y).
Linguagem de consulta: Cypher.

--------------------------------------------------------------------

// 1. Criando os Nós (Nodes)
// Representam as entidades do sistema (Leitores e Obras).
CREATE (u1:Leitor {id: 'user_9921', nome: 'Gustavo'})
CREATE (u2:Leitor {id: 'user_883a', nome: 'Maria'})
CREATE (o1:Obra {id: 'ISBN-9788535914849', titulo: 'Dom Casmurro'})
CREATE (o2:Obra {id: 'AUD-998822', titulo: 'O Hobbit'})
CREATE (o3:Obra {id: 'ISBN-112233', titulo: 'O Senhor dos Anéis'})

// 2. Criando os Relacionamentos (Edges)
// Mapeia a ação de "Ler", guardando a data e a nota dada pelo leitor.
CREATE (u1)-[:LEU {data: '2026-10-15', nota: 5}]->(o1)
CREATE (u1)-[:LEU {data: '2026-11-01', nota: 4}]->(o2)
CREATE (u2)-[:LEU {data: '2026-09-20', nota: 5}]->(o1)
CREATE (u2)-[:LEU {data: '2026-10-05', nota: 5}]->(o3)

// 3. Query de Recomendação (A inteligência do Grafo)
// Busca livros que pessoas com o mesmo gosto do Gustavo leram, 
// mas que ele ainda não leu. Ordena pelos mais recomendados.
MATCH (u1:Leitor {id: 'user_9921'})-[:LEU]->(obra_comum:Obra)<-[:LEU]-(outro_leitor:Leitor)-[:LEU]->(recomendacao:Obra)
WHERE NOT (u1)-[:LEU]->(recomendacao)
RETURN recomendacao.titulo AS Livro_Recomendado, COUNT(outro_leitor) AS Forca_Recomendacao
ORDER BY Forca_Recomendacao DESC 
LIMIT 5;

/*
EXPLICAÇÃO TÉCNICA:
O Neo4j não utiliza tabelas ou JOINs para fazer isso. Ele percorre 
os ponteiros de memória (Edges). Se o Gustavo leu "Dom Casmurro" e 
a Maria também leu "Dom Casmurro", o banco entende que há uma 
conexão (aresta) forte entre eles. Se a Maria leu "O Senhor dos Anéis" 
e o Gustavo não, o sistema recomenda esse livro instantaneamente.
*/
